# FitsCull v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** App nativa macOS che scorre una cartella di FITS astrofotografici, ne mostra anteprima stretchata, statistiche e metriche stellari, e sposta nel Cestino i file scartati.

**Architecture:** Pacchetto SwiftPM con una libreria `FitsCullCore` (lettura FITS, decodifica Bayer, statistiche, rilevamento stelle, stretch, cache, regole di sessione, cestino — tutto testato con Swift Testing), un eseguibile SwiftUI `FitsCull` (interfaccia, scheduling dell'analisi in background) e un eseguibile da riga di comando `fcanalyze` per misure di prestazione e confronto con PixInsight. Uno script assembla `FitsCull.app` e la installa su Dati2.

**Tech Stack:** Swift 6.3 (Command Line Tools 26.6, niente Xcode), SwiftUI, Swift Charts, Accelerate non necessario (loop Swift ottimizzati in release), Swift Testing.

**Spec:** `docs/superpowers/specs/2026-10-11-fitscull-design.md`

## Note rispetto allo spec

- **Scala pixel:** `XPIXSZ` negli header ASIAIR/N.I.N.A. è già comprensivo di binning (convenzione SBIG), quindi la scala è `206,265 × XPIXSZ / FOCALLEN` senza moltiplicare per il binning. Il Task 4 corregge la riga nello spec.
- **Saturazione stelle:** sulla luminanza a metà risoluzione una stella è considerata satura se il picco ≥ 95% del massimo rappresentabile (la media R/G/B attenua la saturazione di un solo canale).
- **Ordinamento:** menu "Ordina" nella barra (Data / FWHM / Stelle / Fondo) invece di intestazioni di colonna cliccabili.
- **`fcanalyze`:** eseguibile da riga di comando aggiunto per misurare i tempi e per la validazione §9 dello spec.
- **Regola FWHM** calcolata in pixel (indipendente dalla scala); grafici e soglie in ″ quando la scala è nota.

## Review Focus

1. **Camera mono / FITS senza `BAYERPAT`** → anteprima in grigi, un solo canale "L", stelle rilevate a piena risoluzione. Test nel Task 9 (`analyzesMonoFile`).
2. **NAS SMB senza Cestino** → i file risultano "Cestino non disponibile", nessuna eliminazione silenziosa, si offre il ripiego. Test nel Task 12 (`reportsTrashUnavailable`).
3. **File troncato (copia interrotta), compresso o cubo nella cartella** → la riga mostra il motivo, gli altri file funzionano. Test nel Task 3 (`truncatedDataIsReported`, `rejectsCubes`, `rejectsEmptyPrimaryHDU`) e nel Task 10 (`scanKeepsBadFilesWithReason`).
4. **Frame nuvoloso senza stelle rilevabili** → metriche stellari assenti (non NaN/crash), segnalato per numero stelle, cache serializzabile. Test nel Task 8 (`emptyFieldHasNoStars`), Task 11 (`zeroStarFrameIsFlaggedNotCrashing`), Task 10 (`nanValuesRoundTrip`).
5. **File sovrascritto con lo stesso nome** (dimensione o data diversa) → la cache non restituisce metriche vecchie. Test nel Task 10 (`modifiedFileMissesCache`).

```swift
import Foundation

/// Costruisce file FITS sintetici per i test.
enum FITSFixture {
    static func card(_ s: String) -> String {
        s.padding(toLength: 80, withPad: " ", startingAt: 0)
    }

    static func kv(_ key: String, _ value: String) -> String {
        key.padding(toLength: 8, withPad: " ", startingAt: 0) + "= " + value
    }

    /// Header completo: card + END, riempito a multipli di 2880 byte.
    static func header(_ cards: [String]) -> Data {
        var text = cards.map(card).joined() + card("END")
        let remainder = text.count % 2880
        if remainder != 0 { text += String(repeating: " ", count: 2880 - remainder) }
        return Data(text.utf8)
    }

    /// File FITS completo. `pixels` sono valori fisici (dopo BZERO/BSCALE), riga per riga.
    static func image(bitpix: Int, width: Int, height: Int, pixels: [Double],
                      bzero: Double = 0, bscale: Double = 1, extra: [String] = []) -> Data {
        var cards = [kv("SIMPLE", "T"), kv("BITPIX", "\(bitpix)"), kv("NAXIS", "2"),
                     kv("NAXIS1", "\(width)"), kv("NAXIS2", "\(height)")]
        if bzero != 0 { cards.append(kv("BZERO", "\(bzero)")) }
        if bscale != 1 { cards.append(kv("BSCALE", "\(bscale)")) }
        cards += extra
        var data = header(cards)
        var body = Data()
        for p in pixels {
            let raw = ((p - bzero) / bscale).rounded()
            switch bitpix {
            case 8:
                body.append(UInt8(max(0, min(255, raw))))
            case 16:
                withUnsafeBytes(of: Int16(max(-32768, min(32767, raw))).bigEndian) { body.append(contentsOf: $0) }
            case 32:
                withUnsafeBytes(of: Int32(raw).bigEndian) { body.append(contentsOf: $0) }
            case -32:
                withUnsafeBytes(of: Float((p - bzero) / bscale).bitPattern.bigEndian) { body.append(contentsOf: $0) }
            case -64:
                withUnsafeBytes(of: ((p - bzero) / bscale).bitPattern.bigEndian) { body.append(contentsOf: $0) }
            default:
                fatalError("BITPIX non gestito nel fixture")
            }
        }
        let remainder = body.count % 2880
        if remainder != 0 { body.append(Data(count: 2880 - remainder)) }
        data.append(body)
        return data
    }

    static func tempDirectory() throws -> URL {
        let dir = FileManager.default.temporaryDirectory
            .appendingPathComponent("FitsCullTests-\(UUID().uuidString)", isDirectory: true)
        try FileManager.default.createDirectory(at: dir, withIntermediateDirectories: true)
        return dir
    }

    @discardableResult
    static func write(_ data: Data, name: String = "frame.fits", in directory: URL? = nil) throws -> URL {
        let dir = try directory ?? tempDirectory()
        let url = dir.appendingPathComponent(name)
        try data.write(to: url)
        return url
    }
}
```

`Sources/FitsCullCore/FITS/FITSHeader.swift`:
```swift
import Foundation

public enum FITSError: Error, Equatable, Sendable, CustomStringConvertible {
    case notFITS
    case truncatedHeader
    case truncatedData
    case unsupported(String)

    public var description: String {
        switch self {
        case .notFITS: "non è un file FITS"
        case .truncatedHeader: "intestazione FITS incompleta"
        case .truncatedData: "file troncato (copia interrotta?)"
        case .unsupported(let reason): "non supportato: \(reason)"
        }
    }
}

public enum FITSValue: Sendable, Equatable, Codable {
    case string(String)
    case int(Int)
    case double(Double)
    case bool(Bool)
}

public struct FITSCard: Sendable, Equatable, Codable {
    public let key: String
    public let value: FITSValue?
    public let comment: String
    public let raw: String

    public init(key: String, value: FITSValue?, comment: String, raw: String) {
        self.key = key
        self.value = value
        self.comment = comment
        self.raw = raw
    }
}

public struct FITSHeader: Sendable, Equatable {
    public static let blockSize = 2880
    public static let cardSize = 80

    public let cards: [FITSCard]
    /// Lunghezza dell'header in byte, compreso il riempimento a multipli di 2880.
    public let byteLength: Int

    public init(cards: [FITSCard], byteLength: Int) {
        self.cards = cards
        self.byteLength = byteLength
    }

    public subscript(key: String) -> FITSValue? {
        cards.first { $0.key == key && $0.value != nil }?.value
    }

    public func string(_ key: String) -> String? {
        switch self[key] {
        case .string(let s): s
        case .int(let i): String(i)
        case .double(let d): String(d)
        case .bool(let b): b ? "T" : "F"
        case nil: nil
        }
    }

    public func double(_ key: String) -> Double? {
        switch self[key] {
        case .double(let d): d
        case .int(let i): Double(i)
        case .string(let s): Double(s)
        default: nil
        }
    }

    public func int(_ key: String) -> Int? {
        switch self[key] {
        case .int(let i): i
        case .double(let d) where d.rounded() == d && abs(d) < 1e15: Int(d)
        case .string(let s): Int(s)
        default: nil
        }
    }

    public func bool(_ key: String) -> Bool? {
        if case .bool(let b) = self[key] { return b }
        return nil
    }

    /// Legge card da 80 byte fino a END. Il primo card deve essere SIMPLE.
    public static func parse(_ data: Data) throws -> FITSHeader {
        var cards: [FITSCard] = []
        var offset = 0
        while offset + cardSize <= data.count {
            let start = data.startIndex + offset
            let card = parseCard(String(decoding: data[start ..< start + cardSize], as: UTF8.self))
            if cards.isEmpty && card.key != "SIMPLE" { throw FITSError.notFITS }
            offset += cardSize
            if card.key == "END" {
                let blocks = (offset + blockSize - 1) / blockSize
                return FITSHeader(cards: cards, byteLength: blocks * blockSize)
            }
            cards.append(card)
        }
        if cards.isEmpty { throw FITSError.notFITS }
        throw FITSError.truncatedHeader
    }

    static func parseCard(_ raw: String) -> FITSCard {
        let bytes = Array(raw.utf8)
        let key = String(decoding: bytes.prefix(8), as: UTF8.self).trimmingCharacters(in: .whitespaces)
        guard bytes.count >= 10, bytes[8] == UInt8(ascii: "="), bytes[9] == UInt8(ascii: " ") else {
            return FITSCard(key: key, value: nil, comment: "", raw: raw)
        }
        let field = Array(bytes[10...])
        var i = 0
        while i < field.count, field[i] == UInt8(ascii: " ") { i += 1 }

        if i < field.count, field[i] == UInt8(ascii: "'") {
            var chars: [UInt8] = []
            i += 1
            while i < field.count {
                if field[i] == UInt8(ascii: "'") {
                    if i + 1 < field.count, field[i + 1] == UInt8(ascii: "'") {
                        chars.append(field[i])
                        i += 2
                        continue
                    }
                    i += 1
                    break
                }
                chars.append(field[i])
                i += 1
            }
            let value = String(decoding: chars, as: UTF8.self).trimmingCharacters(in: .whitespaces)
            return FITSCard(key: key, value: .string(value), comment: comment(in: field, from: i), raw: raw)
        }

        let rest = String(decoding: field[i...], as: UTF8.self)
        let parts = rest.split(separator: "/", maxSplits: 1, omittingEmptySubsequences: false)
        let token = parts.first.map { $0.trimmingCharacters(in: .whitespaces) } ?? ""
        let comment = parts.count > 1 ? parts[1].trimmingCharacters(in: .whitespaces) : ""
        return FITSCard(key: key, value: parseValue(token), comment: comment, raw: raw)
    }

    private static func comment(in field: [UInt8], from index: Int) -> String {
        guard index < field.count else { return "" }
        let rest = String(decoding: field[index...], as: UTF8.self)
        guard let slash = rest.firstIndex(of: "/") else { return "" }
        return rest[rest.index(after: slash)...].trimmingCharacters(in: .whitespaces)
    }

    static func parseValue(_ token: String) -> FITSValue? {
        if token == "T" { return .bool(true) }
        if token == "F" { return .bool(false) }
        if let i = Int(token) { return .int(i) }
        if let d = Double(token.replacingOccurrences(of: "D", with: "E")) { return .double(d) }
        return nil
    }
}
```

```swift
import Foundation

public struct FITSImage: Sendable {
    public let header: FITSHeader
    public let width: Int
    public let height: Int
    public let bitpix: Int
    /// Valori fisici (BZERO/BSCALE applicati), riga per riga; la riga 0 è la prima del file (in alto).
    public let pixels: [Float]
    /// Massimo valore fisico rappresentabile dal tipo intero; nil per dati in virgola mobile.
    public let maxRepresentable: Float?

    public init(header: FITSHeader, width: Int, height: Int, bitpix: Int, pixels: [Float], maxRepresentable: Float?) {
        self.header = header
        self.width = width
        self.height = height
        self.bitpix = bitpix
        self.pixels = pixels
        self.maxRepresentable = maxRepresentable
    }
}

public enum FITSReader {
    /// Legge solo i blocchi dell'header (veloce anche su NAS).
    public static func readHeader(url: URL) throws -> FITSHeader {
        let handle = try FileHandle(forReadingFrom: url)
        defer { try? handle.close() }
        var data = Data()
        for _ in 0..<200 {
            guard let block = try handle.read(upToCount: FITSHeader.blockSize), !block.isEmpty else { break }
            data.append(block)
            do {
                return try FITSHeader.parse(data)
            } catch FITSError.truncatedHeader {
                continue
            }
        }
        throw FITSError.truncatedHeader
    }

    public static func read(url: URL) throws -> FITSImage {
        try decode(Data(contentsOf: url, options: .alwaysMapped))
    }

    public static func decode(_ data: Data) throws -> FITSImage {
        let header = try FITSHeader.parse(data)
        guard header.bool("SIMPLE") == true else { throw FITSError.notFITS }
        let naxis = header.int("NAXIS") ?? 0
        guard naxis != 0 else { throw FITSError.unsupported("nessuna immagine nell'HDU primaria (file compresso?)") }
        guard naxis == 2 else { throw FITSError.unsupported("NAXIS = \(naxis)") }
        guard let bitpix = header.int("BITPIX"), [8, 16, 32, -32].contains(bitpix) else {
            throw FITSError.unsupported("BITPIX = \(header.int("BITPIX").map(String.init) ?? "?")")
        }
        guard let width = header.int("NAXIS1"), let height = header.int("NAXIS2"), width > 0, height > 0 else {
            throw FITSError.unsupported("dimensioni non valide")
        }
        let bzero = header.double("BZERO") ?? 0
        let bscale = header.double("BSCALE") ?? 1
        let count = width * height
        let start = header.byteLength
        guard data.count >= start + count * abs(bitpix) / 8 else { throw FITSError.truncatedData }

        var pixels = [Float](repeating: 0, count: count)
        let scale = Float(bscale)
        let zero = Float(bzero)
        data.withUnsafeBytes { (raw: UnsafeRawBufferPointer) in
            let base = raw.baseAddress!.advanced(by: start)
            pixels.withUnsafeMutableBufferPointer { out in
                switch bitpix {
                case 8:
                    for i in 0..<count {
                        out[i] = Float(base.load(fromByteOffset: i, as: UInt8.self)) * scale + zero
                    }
                case 16:
                    for i in 0..<count {
                        let bits = UInt16(bigEndian: base.loadUnaligned(fromByteOffset: i * 2, as: UInt16.self))
                        out[i] = Float(Int16(bitPattern: bits)) * scale + zero
                    }
                case 32:
                    for i in 0..<count {
                        let bits = UInt32(bigEndian: base.loadUnaligned(fromByteOffset: i * 4, as: UInt32.self))
                        out[i] = Float(Double(Int32(bitPattern: bits)) * bscale + bzero)
                    }
                default:
                    for i in 0..<count {
                        let bits = UInt32(bigEndian: base.loadUnaligned(fromByteOffset: i * 4, as: UInt32.self))
                        out[i] = Float(bitPattern: bits) * scale + zero
                    }
                }
            }
        }
        let maxRaw: Double? = switch bitpix {
        case 8: 255
        case 16: 32767
        case 32: 2_147_483_647
        default: nil
        }
        return FITSImage(header: header, width: width, height: height, bitpix: bitpix, pixels: pixels,
                         maxRepresentable: maxRaw.map { Float($0 * bscale + bzero) })
    }
}
```

```swift
import Foundation

public enum ImageType: String, Sendable, Codable, CaseIterable {
    case light, dark, flat, bias, darkFlat, unknown

    public init(fitsValue: String?) {
        let v = (fitsValue ?? "").lowercased().replacingOccurrences(of: " ", with: "")
        if v.contains("light") { self = .light }
        else if v.contains("darkflat") || v.contains("flatdark") { self = .darkFlat }
        else if v.contains("dark") { self = .dark }
        else if v.contains("flat") { self = .flat }
        else if v.contains("bias") || v.contains("offset") { self = .bias }
        else { self = .unknown }
    }

    public var label: String {
        switch self {
        case .light: "Light"
        case .dark: "Dark"
        case .flat: "Flat"
        case .bias: "Bias"
        case .darkFlat: "Dark flat"
        case .unknown: "Sconosciuto"
        }
    }
}

public struct FrameMeta: Sendable, Codable, Equatable {
    public var width: Int
    public var height: Int
    public var exposure: Double?
    public var gain: Int?
    public var offset: Int?
    public var sensorTemp: Double?
    public var setTemp: Double?
    public var dateObs: Date?
    public var object: String?
    public var imageType: ImageType
    public var bayerPattern: String?
    public var pixelSizeMicrons: Double?
    public var focalLengthMM: Double?
    public var binning: Int
    public var filter: String?
    public var airmass: Double?
    public var instrument: String?

    public init(header: FITSHeader) {
        width = header.int("NAXIS1") ?? 0
        height = header.int("NAXIS2") ?? 0
        exposure = header.double("EXPTIME") ?? header.double("EXPOSURE")
        gain = header.int("GAIN") ?? header.double("GAIN").map { Int($0.rounded()) }
        offset = header.int("OFFSET")
        sensorTemp = header.double("CCD-TEMP")
        setTemp = header.double("SET-TEMP")
        dateObs = header.string("DATE-OBS").flatMap(FrameMeta.parseDate)
        object = header.string("OBJECT").flatMap { $0.isEmpty ? nil : $0 }
        imageType = ImageType(fitsValue: header.string("IMAGETYP") ?? header.string("FRAME"))
        let pattern = header.string("BAYERPAT")?.uppercased()
        bayerPattern = pattern.flatMap { ["RGGB", "BGGR", "GRBG", "GBRG"].contains($0) ? $0 : nil }
        pixelSizeMicrons = header.double("XPIXSZ")
        focalLengthMM = header.double("FOCALLEN")
        binning = header.int("XBINNING") ?? 1
        filter = header.string("FILTER").flatMap { $0.isEmpty ? nil : $0 }
        airmass = header.double("AIRMASS")
        instrument = header.string("INSTRUME")
    }

    /// Secondi d'arco per pixel del sensore. XPIXSZ include già il binning (convenzione SBIG/ASIAIR/N.I.N.A.).
    public var pixelScale: Double? {
        guard let p = pixelSizeMicrons, let f = focalLengthMM, p > 0, f > 0 else { return nil }
        return 206.264806 * p / f
    }

    /// ISO 8601 senza fuso (UTC), con frazioni di secondo di lunghezza variabile.
    static func parseDate(_ text: String) -> Date? {
        let parts = text.split(separator: ".", maxSplits: 1)
        let formatter = DateFormatter()
        formatter.locale = Locale(identifier: "en_US_POSIX")
        formatter.timeZone = TimeZone(identifier: "UTC")
        formatter.dateFormat = "yyyy-MM-dd'T'HH:mm:ss"
        guard let first = parts.first, let base = formatter.date(from: String(first)) else { return nil }
        guard parts.count == 2, let fraction = Double("0." + parts[1].filter(\.isNumber)) else { return base }
        return base.addingTimeInterval(fraction)
    }
}
```

```swift
    public let height: Int

    public let binFactor: Int

        self.height = height
        self.binFactor = binFactor
        self.planes = planes
    }

    public func luminance() -> [Float] {
        guard isColor else { return planes[0] }
        var out = [Float](repeating: 0, count: width * height)
        let r = planes[0], g = planes[1], b = planes[2]
        for i in 0..<out.count { out[i] = (r[i] + g[i] + b[i]) / 3 }
        return out
    }

public enum FrameDecoder {
    /// Superpixel 2×2 (R, media dei due G, B) a metà risoluzione; mono invariato.
    public static func decode(_ image: FITSImage, pattern: BayerPattern?) -> PlanarImage {
        guard let pattern else {
            return PlanarImage(width: image.width, height: image.height, binFactor: 1, planes: [image.pixels])
        }
        let w = image.width / 2, h = image.height / 2
        let o = pattern.offsets(rowStride: image.width)
        var r = [Float](repeating: 0, count: w * h)
        var g = r
        var b = r
        image.pixels.withUnsafeBufferPointer { src in
            for y in 0..<h {
                for x in 0..<w {
                    let base = 2 * y * image.width + 2 * x
                    let i = y * w + x
                    r[i] = src[base + o.r]
                    g[i] = (src[base + o.g1] + src[base + o.g2]) * 0.5
                    b[i] = src[base + o.b]
                }
            }
        }
        return PlanarImage(width: w, height: h, binFactor: 2, planes: [r, g, b])
    }

    /// Campioni CFA grezzi per colore, per le statistiche: [R, G, B] (G contiene entrambi i verdi) o [L].
    public static func cfaChannels(_ image: FITSImage, pattern: BayerPattern?) -> [[Float]] {
        guard let pattern else { return [image.pixels] }
        let w = image.width / 2, h = image.height / 2
        let o = pattern.offsets(rowStride: image.width)
        var r = [Float](), g = [Float](), b = [Float]()
        r.reserveCapacity(w * h)
        g.reserveCapacity(2 * w * h)
        b.reserveCapacity(w * h)

            for y in 0..<h {
                for x in 0..<w {

                    r.append(src[base + o.r])
                    g.append(src[base + o.g1])
                    g.append(src[base + o.g2])
                    b.append(src[base + o.b])
                }
            }
        }
        return [r, g, b]
    }
```

```swift
/// Mediana e MAD in O(n) con istogramma a 65536 classi: esatte per dati interi a 16 bit,
/// risoluzione (max − min) / 65535 negli altri casi. I valori non finiti sono ignorati.
public enum RobustStats {
    public static let histogramBins = 65536

    public static func finiteRange(_ v: [Float]) -> (lo: Float, hi: Float, count: Int)? {
        var lo = Float.infinity, hi = -Float.infinity, n = 0
        for x in v where x.isFinite {
            lo = min(lo, x)
            hi = max(hi, x)
            n += 1
        }
        return n == 0 ? nil : (lo, hi, n)
    }

    public static func median(_ v: [Float], lo: Float, hi: Float) -> Double {
        guard hi > lo else { return Double(lo) }
        let width = (hi - lo) / Float(histogramBins - 1)
        var hist = [Int](repeating: 0, count: histogramBins)
        var n = 0
        for x in v where x.isFinite {
            let i = Int(((x - lo) / width).rounded())
            hist[min(max(i, 0), histogramBins - 1)] += 1
            n += 1
        }
        return Double(lo) + Double(lowerMedianBin(hist, count: n)) * Double(width)
    }

    public static func mad(_ v: [Float], median: Double) -> Double {
        let m = Float(median)
        var maxDev: Float = 0
        var n = 0
        for x in v where x.isFinite {
            maxDev = max(maxDev, abs(x - m))
            n += 1
        }
        guard maxDev > 0, n > 0 else { return 0 }
        let width = maxDev / Float(histogramBins - 1)
        var hist = [Int](repeating: 0, count: histogramBins)
        for x in v where x.isFinite {
            hist[min(Int((abs(x - m) / width).rounded()), histogramBins - 1)] += 1
        }
        return Double(lowerMedianBin(hist, count: n)) * Double(width)
    }

    public static func medianAndMAD(_ v: [Float]) -> (median: Double, mad: Double) {
        guard let range = finiteRange(v) else { return (.nan, .nan) }
        let m = median(v, lo: range.lo, hi: range.hi)
        return (m, mad(v, median: m))
    }

    /// Mediana esatta (media dei due centrali per n pari), per vettori piccoli.
    public static func exactMedian(_ v: [Double]) -> Double {
        let s = v.filter(\.isFinite).sorted()
        guard !s.isEmpty else { return .nan }
        return s.count % 2 == 1 ? s[s.count / 2] : (s[s.count / 2 - 1] + s[s.count / 2]) / 2
    }

    static func lowerMedianBin(_ hist: [Int], count: Int) -> Int {
        let target = (count + 1) / 2
        var accumulated = 0
        for i in hist.indices {
            accumulated += hist[i]
            if accumulated >= target { return i }
        }
        return hist.count - 1
    }
}
```

```swift
public struct ChannelStats: Sendable, Codable, Equatable {
    public var min: Double
    public var max: Double
    public var mean: Double
    public var median: Double
    public var stdDev: Double
    /// 1,4826 × MAD: stima robusta del rumore.
    public var noise: Double
    /// Frazione (0…1) di pixel ≥ 99,5% del massimo rappresentabile; nil se il livello non è noto.
    public var saturatedFraction: Double?

    public init(min: Double, max: Double, mean: Double, median: Double, stdDev: Double, noise: Double,
                saturatedFraction: Double?) {
        self.min = min
        self.max = max
        self.mean = mean
        self.median = median
        self.stdDev = stdDev
        self.noise = noise
        self.saturatedFraction = saturatedFraction
    }
}

public enum Statistics {
    public static let saturationRatio: Float = 0.995

    public static func channel(_ v: [Float], saturationLevel: Float?) -> ChannelStats {
        guard let range = RobustStats.finiteRange(v) else {
            return ChannelStats(min: .nan, max: .nan, mean: .nan, median: .nan, stdDev: .nan, noise: .nan,
                                saturatedFraction: nil)
        }
        let threshold = saturationLevel.map { $0 * saturationRatio }
        var sum = 0.0, sumSquares = 0.0
        var saturated = 0
        for x in v where x.isFinite {
            let d = Double(x)
            sum += d
            sumSquares += d * d
            if let t = threshold, x >= t { saturated += 1 }
        }
        let n = Double(range.count)
        let mean = sum / n
        let variance = Swift.max(0, sumSquares / n - mean * mean)
        let median = RobustStats.median(v, lo: range.lo, hi: range.hi)
        let mad = RobustStats.mad(v, median: median)
        return ChannelStats(min: Double(range.lo), max: Double(range.hi), mean: mean, median: median,
                            stdDev: variance.squareRoot(), noise: 1.4826 * mad,
                            saturatedFraction: saturationLevel == nil ? nil : Double(saturated) / n)
    }
}
```

```swift
struct TestRandom {
    private var state: UInt64

    init(seed: UInt64) { state = seed }

    mutating func next() -> UInt64 {
        state &+= 0x9E37_79B9_7F4A_7C15
        var z = state
        z = (z ^ (z >> 30)) &* 0xBF58_476D_1CE4_E5B9
        z = (z ^ (z >> 27)) &* 0x94D0_49BB_1331_11EB
        return z ^ (z >> 31)
    }

    mutating func uniform() -> Double { Double(next() >> 11) / Double(1 << 53) }

    mutating func gaussian(mean: Double = 0, sigma: Double = 1) -> Double {
        let u1 = max(uniform(), 1e-12), u2 = uniform()
        return mean + sigma * (-2 * log(u1)).squareRoot() * cos(2 * .pi * u2)
```

```swift
import CoreGraphics
import Foundation

public struct StretchParams: Sendable, Equatable {
    public var shadows: Double
    public var midtones: Double

    public init(shadows: Double, midtones: Double) {
        self.shadows = shadows
        self.midtones = midtones
    }

    public static let identity = StretchParams(shadows: 0, midtones: 0.5)
}

public struct StretchedImage: Sendable {
    public let width: Int
    public let height: Int
    /// RGBA 8 bit, alfa sempre 255.
    public let rgba: [UInt8]
}

/// Stretch automatico come l'AutoSTF di PixInsight.
public enum AutoStretch {
    public static let targetBackground = 0.25
    public static let shadowsClipping = -2.8

    /// Midtones transfer function.
    public static func mtf(_ m: Double, _ x: Double) -> Double {
        if x <= 0 { return 0 }
        if x >= 1 { return 1 }
        return ((m - 1) * x) / ((2 * m - 1) * x - m)
    }

    /// Parametri da mediana e MAD normalizzati in [0, 1].
    public static func params(median: Double, mad: Double) -> StretchParams {
        let shadows = min(max(median + shadowsClipping * 1.4826 * mad, 0), 1)
        guard median > shadows, shadows < 1 else { return .identity }
        let x0 = (median - shadows) / (1 - shadows)
        return StretchParams(shadows: shadows, midtones: mtf(targetBackground, x0))
    }

    public static func stretch(_ image: PlanarImage, normalization: Float, linked: Bool, enabled: Bool) -> StretchedImage {
        let n = image.width * image.height
        let norm = normalization > 0 ? normalization : 1
        var params = Array(repeating: StretchParams.identity, count: image.planes.count)
        if enabled {
            let measured = image.planes.map { plane -> (median: Double, mad: Double) in
                let m = RobustStats.medianAndMAD(plane)
                return (m.median / Double(norm), m.mad / Double(norm))
            }
            if linked {
                let median = measured.map(\.median).reduce(0, +) / Double(measured.count)
                let mad = measured.map(\.mad).reduce(0, +) / Double(measured.count)
                params = Array(repeating: Self.params(median: median, mad: mad), count: measured.count)
            } else {
                params = measured.map { Self.params(median: $0.median, mad: $0.mad) }
            }
        }
        let tables = params.map(lookupTable)
        var rgba = [UInt8](repeating: 255, count: n * 4)
        rgba.withUnsafeMutableBufferPointer { out in
            for (c, plane) in image.planes.enumerated() {
                let table = tables[c]
                plane.withUnsafeBufferPointer { p in
                    for i in 0..<n {
                        let x = p[i].isFinite ? min(max(p[i] / norm, 0), 1) : 0
                        let v = table[Int(x * 65535)]
                        if image.isColor {
                            out[i * 4 + c] = v
                        } else {
                            out[i * 4] = v
                            out[i * 4 + 1] = v
                            out[i * 4 + 2] = v
                        }
                    }
                }
            }
        }
        return StretchedImage(width: image.width, height: image.height, rgba: rgba)
    }

    static func lookupTable(_ p: StretchParams) -> [UInt8] {
        (0..<65536).map { i in
            let x = Double(i) / 65535
            let y = x <= p.shadows ? 0 : mtf(p.midtones, (x - p.shadows) / (1 - p.shadows))
            return UInt8((y * 255).rounded())
        }
    }

    public static func cgImage(_ s: StretchedImage) -> CGImage? {
        guard let provider = CGDataProvider(data: Data(s.rgba) as CFData),
              let space = CGColorSpace(name: CGColorSpace.sRGB) else { return nil }
        return CGImage(width: s.width, height: s.height, bitsPerComponent: 8, bitsPerPixel: 32,
                       bytesPerRow: s.width * 4, space: space,
                       bitmapInfo: CGBitmapInfo(rawValue: CGImageAlphaInfo.noneSkipLast.rawValue),
                       provider: provider, decode: nil, shouldInterpolate: true, intent: .defaultIntent)
    }
}

public struct DisplayHistogram: Sendable, Equatable {
    /// 256 classi per canale visualizzato (3 a colori, 1 mono).
    public let channels: [[Int]]

    public init(_ s: StretchedImage, isColor: Bool) {
        let count = isColor ? 3 : 1
        var bins = Array(repeating: [Int](repeating: 0, count: 256), count: count)
        let n = s.width * s.height
        for i in 0..<n {
            for c in 0..<count { bins[c][Int(s.rgba[i * 4 + c])] += 1 }
        }
        channels = bins
    }
}
```

```swift
    var x: Double
    var y: Double
    var amplitude: Double
    var sigmaX: Double
    var sigmaY: Double
}

/// Campi stellari sintetici: fondo + rumore gaussiano + stelle gaussiane.
enum SyntheticField {
    static func make(width: Int, height: Int, background: (Int, Int) -> Double = { _, _ in 1000 },
                     noise: Double = 10, stars: [SyntheticStar], seed: UInt64 = 1, clip: Double? = nil) -> [Float] {
        var rng = TestRandom(seed: seed)
        var v = [Double](repeating: 0, count: width * height)
        for y in 0..<height {
            for x in 0..<width { v[y * width + x] = background(x, y) + rng.gaussian(sigma: noise) }
        }
        for s in stars {
            let rx = Int(6 * s.sigmaX) + 1, ry = Int(6 * s.sigmaY) + 1
            for y in max(0, Int(s.y) - ry)...min(height - 1, Int(s.y) + ry) {
                for x in max(0, Int(s.x) - rx)...min(width - 1, Int(s.x) + rx) {
                    let dx = Double(x) - s.x, dy = Double(y) - s.y
                    v[y * width + x] += s.amplitude * exp(-(dx * dx / (2 * s.sigmaX * s.sigmaX) + dy * dy / (2 * s.sigmaY * s.sigmaY)))
                }
            }
        }
        if let clip { v = v.map { min($0, clip) } }
        return v.map(Float.init)
    }

    /// Griglia 8×8 di stelle, distanti 56 px, a partire da (60, 60): adatta a un campo 512×512.
    static func grid(amplitude: Double = 2000, sigmaX: Double = 2, sigmaY: Double = 2) -> [SyntheticStar] {
        (0..<8).flatMap { j in
            (0..<8).map { i in
                SyntheticStar(x: 60 + 56 * Double(i) + 0.3, y: 60 + 56 * Double(j) + 0.3,
                              amplitude: amplitude, sigmaX: sigmaX, sigmaY: sigmaY)
            }
        }
    }
}
```

```swift
import Foundation

public struct StarMetrics: Sendable, Codable, Equatable {
    public var count: Int
    /// Half-flux radius mediano, in pixel dell'immagine analizzata.
    public var hfr: Double?
    /// FWHM mediana da momenti secondi, in pixel dell'immagine analizzata.
    public var fwhm: Double?
    public var fwhmArcsec: Double?
    public var eccentricity: Double?

    public init(count: Int, hfr: Double?, fwhm: Double?, fwhmArcsec: Double?, eccentricity: Double?) {
        self.count = count
        self.hfr = hfr
        self.fwhm = fwhm
        self.fwhmArcsec = fwhmArcsec
        self.eccentricity = eccentricity
    }

    public static let none = StarMetrics(count: 0, hfr: nil, fwhm: nil, fwhmArcsec: nil, eccentricity: nil)
}

public struct DetectedStar: Sendable, Equatable {
    public var x: Double
    public var y: Double
    public var flux: Double
    public var hfr: Double
    public var fwhm: Double
    public var eccentricity: Double
}

public enum StarDetector {
    public static let tileSize = 64
    public static let sigmaThreshold: Float = 5
    public static let minArea = 3
    public static let maxArea = 2000
    /// Sulla luminanza (media R/G/B) una stella è satura sopra il 95% del massimo rappresentabile.
    public static let saturationRatio: Float = 0.95

    public static func detect(_ v: [Float], width: Int, height: Int, saturationLevel: Float?,
                              arcsecPerPixel: Double?) -> StarMetrics {
        let stars = findStars(v, width: width, height: height, saturationLevel: saturationLevel)
        guard !stars.isEmpty else { return .none }
        let fwhm = RobustStats.exactMedian(stars.map(\.fwhm))
        return StarMetrics(count: stars.count,
                           hfr: RobustStats.exactMedian(stars.map(\.hfr)),
                           fwhm: fwhm,
                           fwhmArcsec: arcsecPerPixel.map { $0 * fwhm },
                           eccentricity: RobustStats.exactMedian(stars.map(\.eccentricity)))
    }

    public static func findStars(_ v: [Float], width: Int, height: Int, saturationLevel: Float?) -> [DetectedStar] {
        guard width >= 16, height >= 16, v.count == width * height else { return [] }
        let background = backgroundMap(v, width: width, height: height)
        var residual = [Float](repeating: 0, count: v.count)
        for i in 0..<v.count { residual[i] = v[i].isFinite ? v[i] - background[i] : 0 }
        let sigma = Float(1.4826 * RobustStats.medianAndMAD(residual).mad)
        guard sigma > 0, sigma.isFinite else { return [] }
        let threshold = sigmaThreshold * sigma

        var visited = [Bool](repeating: false, count: v.count)
        var stack: [Int] = []
        var stars: [DetectedStar] = []
        for start in 0..<v.count where !visited[start] && residual[start] > threshold {
            visited[start] = true
            stack.append(start)
            var area = 0
            var peak = start
            var touchesBorder = false
            while let i = stack.popLast() {
                area += 1
                if residual[i] > residual[peak] { peak = i }
                let x = i % width, y = i / width
                if x == 0 || y == 0 || x == width - 1 || y == height - 1 { touchesBorder = true }
                for dy in -1...1 {
                    for dx in -1...1 where dx != 0 || dy != 0 {
                        let nx = x + dx, ny = y + dy
                        guard nx >= 0, ny >= 0, nx < width, ny < height else { continue }
                        let j = ny * width + nx
                        if !visited[j] && residual[j] > threshold {
                            visited[j] = true
                            stack.append(j)
                        }
                    }
                }
            }
            guard area >= minArea, area <= maxArea, !touchesBorder else { continue }
            if let level = saturationLevel, v[peak] >= level * saturationRatio { continue }
            if let star = measure(residual, width: width, height: height, peak: peak, area: area) {
                stars.append(star)
            }
        }
        return stars
    }

    /// Centroide, momenti secondi (FWHM, eccentricità) e half-flux radius su una finestra circolare.
    static func measure(_ r: [Float], width: Int, height: Int, peak: Int, area: Int) -> DetectedStar? {
        let px = peak % width, py = peak / width
        let radius = max(3.0, 2.5 * (Double(area) / .pi).squareRoot())
        let reach = Int(radius.rounded(.up))
        guard px - reach >= 0, py - reach >= 0, px + reach < width, py + reach < height else { return nil }
        let radius2 = radius * radius

        var flux = 0.0, sx = 0.0, sy = 0.0
        for y in (py - reach)...(py + reach) {
            for x in (px - reach)...(px + reach) {
                let dx = Double(x - px), dy = Double(y - py)
                guard dx * dx + dy * dy <= radius2 else { continue }
                let w = Double(r[y * width + x])
                flux += w
                sx += w * Double(x)
                sy += w * Double(y)
            }
        }
        guard flux > 0 else { return nil }
        let cx = sx / flux, cy = sy / flux

        var total = 0.0, mxx = 0.0, myy = 0.0, mxy = 0.0
        var profile: [(d: Double, w: Double)] = []
        for y in (py - reach)...(py + reach) {
            for x in (px - reach)...(px + reach) {
                let dx = Double(x) - cx, dy = Double(y) - cy
                let d2 = dx * dx + dy * dy
                guard d2 <= radius2 else { continue }
                let w = Double(r[y * width + x])
                total += w
                mxx += w * dx * dx
                myy += w * dy * dy
                mxy += w * dx * dy
                profile.append((d2.squareRoot(), w))
            }
        }
        guard total > 0 else { return nil }
        mxx /= total
        myy /= total
        mxy /= total
        let half = (mxx + myy) / 2
        let spread = (((mxx - myy) / 2) * ((mxx - myy) / 2) + mxy * mxy).squareRoot()
        let major = half + spread, minor = half - spread
        guard major > 0, minor > 0 else { return nil }

        profile.sort { $0.d < $1.d }
        var accumulated = 0.0, previousDistance = 0.0
        var hfr = profile.last?.d ?? 0
        for p in profile {
            if accumulated + p.w >= total / 2, p.w > 0 {
                hfr = previousDistance + (total / 2 - accumulated) / p.w * (p.d - previousDistance)
                break
            }
            accumulated += p.w
            previousDistance = p.d
        }
        return DetectedStar(x: cx, y: cy, flux: total, hfr: hfr, fwhm: 2.3548 * half.squareRoot(),
                            eccentricity: (1 - minor / major).squareRoot())
    }

    /// Fondo per tasselli (mediana campionata a passo 2) con interpolazione/estrapolazione bilineare.
    static func backgroundMap(_ v: [Float], width: Int, height: Int) -> [Float] {
        let gx = max(1, width / tileSize), gy = max(1, height / tileSize)
        let tw = width / gx, th = height / gy
        var grid = [Float](repeating: 0, count: gx * gy)
        var samples: [Float] = []
        for ty in 0..<gy {
            for tx in 0..<gx {
                samples.removeAll(keepingCapacity: true)
                let x0 = tx * tw, y0 = ty * th
                let x1 = tx == gx - 1 ? width : x0 + tw
                let y1 = ty == gy - 1 ? height : y0 + th
                for y in stride(from: y0, to: y1, by: 2) {
                    for x in stride(from: x0, to: x1, by: 2) {
                        let value = v[y * width + x]
                        if value.isFinite { samples.append(value) }
                    }
                }
                samples.sort()
                grid[ty * gx + tx] = samples.isEmpty ? 0 : samples[samples.count / 2]
            }
        }

        func axis(_ p: Int, tile: Int, count: Int) -> (i0: Int, i1: Int, w: Float) {
            guard count > 1 else { return (0, 0, 0) }
            let f = (Double(p) + 0.5) / Double(tile) - 0.5
            let i0 = min(max(Int(f.rounded(.down)), 0), count - 2)
            return (i0, i0 + 1, Float(f - Double(i0)))
        }
        let columns = (0..<width).map { axis($0, tile: tw, count: gx) }
        var out = [Float](repeating: 0, count: v.count)
        for y in 0..<height {
            let row = axis(y, tile: th, count: gy)
            for x in 0..<width {
                let col = columns[x]
                let top = grid[row.i0 * gx + col.i0] * (1 - col.w) + grid[row.i0 * gx + col.i1] * col.w
                let bottom = grid[row.i1 * gx + col.i0] * (1 - col.w) + grid[row.i1 * gx + col.i1] * col.w
                out[y * width + x] = top * (1 - row.w) + bottom * row.w
            }
        }
        return out
    }
}
```

```swift
import Foundation

public struct FrameAnalysis: Sendable, Codable, Equatable {
    public var meta: FrameMeta
    /// [R, G, B] per immagini Bayer, [L] per mono.
    public var channels: [ChannelStats]
    public var stars: StarMetrics

    public init(meta: FrameMeta, channels: [ChannelStats], stars: StarMetrics) {
        self.meta = meta
        self.channels = channels
        self.stars = stars
    }

    /// Fondo cielo: media delle mediane dei canali.
    public var background: Double {
        channels.map(\.median).reduce(0, +) / Double(max(channels.count, 1))
    }

    public var noise: Double {
        channels.map(\.noise).reduce(0, +) / Double(max(channels.count, 1))
    }

    public var saturatedFraction: Double? { channels.compactMap(\.saturatedFraction).max() }

    public var channelNames: [String] { channels.count == 3 ? ["R", "G", "B"] : ["L"] }
}

public struct PreviewSource: Sendable {
    public let image: PlanarImage
    /// Valore che corrisponde al bianco (massimo rappresentabile o massimo dei dati).
    public let normalization: Float
}

public enum FrameAnalyzer {
    public static func analyze(url: URL, includePreview: Bool) throws -> (analysis: FrameAnalysis, preview: PreviewSource?) {
        analyze(try FITSReader.read(url: url), includePreview: includePreview)
    }

    public static func analyze(_ fits: FITSImage, includePreview: Bool) -> (analysis: FrameAnalysis, preview: PreviewSource?) {
        let meta = FrameMeta(header: fits.header)
        let pattern = meta.bayerPattern.flatMap(BayerPattern.init(rawValue:))
        let channels = FrameDecoder.cfaChannels(fits, pattern: pattern)
            .map { Statistics.channel($0, saturationLevel: fits.maxRepresentable) }
        let planar = FrameDecoder.decode(fits, pattern: pattern)
        let stars = StarDetector.detect(planar.luminance(), width: planar.width, height: planar.height,
                                        saturationLevel: fits.maxRepresentable,
                                        arcsecPerPixel: meta.pixelScale.map { $0 * Double(planar.binFactor) })
        let analysis = FrameAnalysis(meta: meta, channels: channels, stars: stars)
        return (analysis, includePreview ? PreviewSource(image: planar, normalization: normalization(fits)) : nil)
    }

    /// Solo decodifica, per file già analizzati (metriche in cache).
    public static func preview(url: URL) throws -> PreviewSource {
        let fits = try FITSReader.read(url: url)
        let pattern = FrameMeta(header: fits.header).bayerPattern.flatMap(BayerPattern.init(rawValue:))
        return PreviewSource(image: FrameDecoder.decode(fits, pattern: pattern), normalization: normalization(fits))
    }

    static func normalization(_ fits: FITSImage) -> Float {
        if let m = fits.maxRepresentable, m > 0 { return m }
        return RobustStats.finiteRange(fits.pixels)?.hi ?? 1
    }
}
```

```swift
import FitsCullCore
import Foundation

// Uso: fcanalyze <file.fits | cartella> …   (compilare con -c release per tempi realistici)
let arguments = CommandLine.arguments.dropFirst()
guard !arguments.isEmpty else {
    print("Uso: fcanalyze <file.fits|cartella> …")
    exit(1)
}

let extensions: Set<String> = ["fit", "fits", "fts"]
var urls: [URL] = []
for argument in arguments {
    let url = URL(fileURLWithPath: argument)
    var isDirectory: ObjCBool = false
    guard FileManager.default.fileExists(atPath: argument, isDirectory: &isDirectory) else {
        print("Non trovato: \(argument)")
        continue
    }
    if isDirectory.boolValue {
        let items = (try? FileManager.default.contentsOfDirectory(at: url, includingPropertiesForKeys: nil)) ?? []
        urls += items.filter { extensions.contains($0.pathExtension.lowercased()) }
            .sorted { $0.lastPathComponent < $1.lastPathComponent }
    } else {
        urls.append(url)
    }
}

func format(_ value: Double?, _ digits: Int) -> String {
    guard let value, value.isFinite else { return "-" }
    return String(format: "%.\(digits)f", value)
}

print(["file", "tipo", "esp", "fondo", "rumore", "sat%", "stelle", "FWHMpx", "FWHM\"", "ecc", "ms"].joined(separator: "\t"))
for url in urls {
    let start = Date()
    do {
        let (a, _) = try FrameAnalyzer.analyze(url: url, includePreview: false)
        let ms = Int(Date().timeIntervalSince(start) * 1000)
        print([url.lastPathComponent, a.meta.imageType.label, format(a.meta.exposure, 0), format(a.background, 1),
               format(a.noise, 1), format(a.saturatedFraction.map { $0 * 100 }, 3), "\(a.stars.count)",
               format(a.stars.fwhm, 2), format(a.stars.fwhmArcsec, 2), format(a.stars.eccentricity, 2), "\(ms)"]
            .joined(separator: "\t"))
    } catch {
        let message = (error as? FITSError)?.description ?? error.localizedDescription
        print("\(url.lastPathComponent)\tERRORE: \(message)")
    }
}
```

```swift
public struct FrameFile: Sendable, Hashable, Codable {
    public let url: URL
    public let size: Int64
    public let modified: Date

    public init(url: URL, size: Int64, modified: Date) {
        self.url = url
        self.size = size
        self.modified = modified

public struct ScannedFrame: Sendable, Identifiable {
    public var id: URL { file.url }
    public let file: FrameFile
    public let header: FITSHeader?
    public let meta: FrameMeta?
    /// Motivo leggibile se l'header non è leggibile.
    public let headerError: String?
}

public enum UserMessage {
    public static func from(_ error: Error) -> String {
        (error as? FITSError)?.description ?? error.localizedDescription
    }
}

public enum FolderScanner {
    public static let extensions: Set<String> = ["fit", "fits", "fts"]

    /// File FITS della cartella (non ricorsivo, senza nascosti), ordinati per nome.
    public static func listFiles(in folder: URL) throws -> [FrameFile] {
        let keys: [URLResourceKey] = [.fileSizeKey, .contentModificationDateKey, .isRegularFileKey]
        let items = try FileManager.default.contentsOfDirectory(at: folder, includingPropertiesForKeys: keys,
                                                                options: [.skipsHiddenFiles])
        return try items.compactMap { url -> FrameFile? in
            guard extensions.contains(url.pathExtension.lowercased()) else { return nil }
            let values = try url.resourceValues(forKeys: Set(keys))
            guard values.isRegularFile == true else { return nil }
            return FrameFile(url: url, size: Int64(values.fileSize ?? 0),
                             modified: values.contentModificationDate ?? .distantPast)
        }
        .sorted { $0.url.lastPathComponent.localizedStandardCompare($1.url.lastPathComponent) == .orderedAscending }
    }

    /// Elenco con header letti (solo i primi blocchi di ogni file), in ordine cronologico.
    public static func scan(_ folder: URL) throws -> [ScannedFrame] {
        try listFiles(in: folder).map { file in
            do {
                let header = try FITSReader.readHeader(url: file.url)
                return ScannedFrame(file: file, header: header, meta: FrameMeta(header: header), headerError: nil)
            } catch {
                return ScannedFrame(file: file, header: nil, meta: nil, headerError: UserMessage.from(error))
            }
        }
        .sorted(by: chronological)
    }

    static func chronological(_ a: ScannedFrame, _ b: ScannedFrame) -> Bool {
        if let da = a.meta?.dateObs, let db = b.meta?.dateObs, da != db { return da < db }
        return a.file.url.lastPathComponent.localizedStandardCompare(b.file.url.lastPathComponent) == .orderedAscending
    }
}
```

```swift
import Foundation

/// Metriche già calcolate, salvate in JSON; chiave = percorso, valide solo se dimensione e data coincidono.
public final class MetricsCache: @unchecked Sendable {
    struct Entry: Codable {
        let size: Int64
        let modified: Double
        let analysis: FrameAnalysis
    }

    private let fileURL: URL
    private let lock = NSLock()
    private var entries: [String: Entry] = [:]
    private var dirty = false

    public init(directory: URL) {
        fileURL = directory.appendingPathComponent("metrics-v1.json")
        if let data = try? Data(contentsOf: fileURL),
           let decoded = try? Self.makeDecoder().decode([String: Entry].self, from: data) {
            entries = decoded
        }
    }

    public static func defaultDirectory() -> URL {
        FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("FitsCull", isDirectory: true)
    }

    public func lookup(_ file: FrameFile) -> FrameAnalysis? {
        lock.withLock {
            guard let entry = entries[file.url.path], entry.size == file.size,
                  entry.modified == file.modified.timeIntervalSince1970 else { return nil }
            return entry.analysis
        }
    }

    public func store(_ analysis: FrameAnalysis, for file: FrameFile) {
        lock.withLock {
            entries[file.url.path] = Entry(size: file.size, modified: file.modified.timeIntervalSince1970, analysis: analysis)
            dirty = true
        }
    }

    public func remove(_ url: URL) {
        lock.withLock {
            entries[url.path] = nil
            dirty = true
        }
    }

    /// Scrive su disco solo se ci sono modifiche.
    public func save() throws {
        let snapshot: [String: Entry]? = lock.withLock { dirty ? entries : nil }
        guard let snapshot else { return }
        try FileManager.default.createDirectory(at: fileURL.deletingLastPathComponent(), withIntermediateDirectories: true)
        try Self.makeEncoder().encode(snapshot).write(to: fileURL, options: .atomic)
        lock.withLock { dirty = false }
    }

    static func makeEncoder() -> JSONEncoder {
        let encoder = JSONEncoder()
        encoder.nonConformingFloatEncodingStrategy = .convertToString(positiveInfinity: "inf", negativeInfinity: "-inf", nan: "nan")
        return encoder
    }

    static func makeDecoder() -> JSONDecoder {
        let decoder = JSONDecoder()
        decoder.nonConformingFloatDecodingStrategy = .convertFromString(positiveInfinity: "inf", negativeInfinity: "-inf", nan: "nan")
        return decoder
    }
}
```

```swift
public struct SessionSummary: Sendable, Equatable {
    public var medianBackground: Double
    public var medianStars: Double
    public var medianFWHM: Double?
    /// 1,4826 × MAD delle FWHM (pixel).
    public var sigmaFWHM: Double?
}

public struct OutlierFlags: OptionSet, Sendable, Hashable {
    public let rawValue: Int
    public init(rawValue: Int) { self.rawValue = rawValue }

    public static let background = OutlierFlags(rawValue: 1 << 0)
    public static let stars = OutlierFlags(rawValue: 1 << 1)
    public static let fwhm = OutlierFlags(rawValue: 1 << 2)
    public static let eccentricity = OutlierFlags(rawValue: 1 << 3)
}

/// Segnalazioni rispetto alla mediana della sessione (spec §5.1). Non scartano nulla da sole.
public enum OutlierRules {
    public static let backgroundRatio = 1.30
    public static let starsRatio = 0.50
    public static let fwhmSigmas = 2.0
    public static let maxEccentricity = 0.6

    public static func summary(_ analyses: [FrameAnalysis]) -> SessionSummary? {
        guard !analyses.isEmpty else { return nil }
        let fwhms = analyses.compactMap(\.stars.fwhm)
        let medianFWHM = fwhms.isEmpty ? nil : RobustStats.exactMedian(fwhms)
        let sigmaFWHM = medianFWHM.map { m in 1.4826 * RobustStats.exactMedian(fwhms.map { abs($0 - m) }) }
        return SessionSummary(medianBackground: RobustStats.exactMedian(analyses.map(\.background)),
                              medianStars: RobustStats.exactMedian(analyses.map { Double($0.stars.count) }),
                              medianFWHM: medianFWHM, sigmaFWHM: sigmaFWHM)
    }

    public static func flags(for a: FrameAnalysis, in s: SessionSummary) -> OutlierFlags {
        var flags: OutlierFlags = []
        if s.medianBackground > 0, a.background > s.medianBackground * backgroundRatio { flags.insert(.background) }
        if s.medianStars > 0, Double(a.stars.count) < s.medianStars * starsRatio { flags.insert(.stars) }
        if let m = s.medianFWHM, let sd = s.sigmaFWHM, let v = a.stars.fwhm,
           v > m + fwhmSigmas * max(sd, 0.05 * m) { flags.insert(.fwhm) }
        if let e = a.stars.eccentricity, e > maxEccentricity { flags.insert(.eccentricity) }
        return flags
    }
}

/// Soglie impostate dall'utente nella vista Sessione (nil = nessuna soglia).
public struct Thresholds: Sendable, Equatable, Codable {
    /// In ″ quando la scala è nota, altrimenti in pixel.
    public var maxFWHM: Double?
    public var minStars: Int?
    public var maxBackground: Double?
    public var maxEccentricity: Double?

    public init(maxFWHM: Double? = nil, minStars: Int? = nil, maxBackground: Double? = nil, maxEccentricity: Double? = nil) {
        self.maxFWHM = maxFWHM
        self.minStars = minStars
        self.maxBackground = maxBackground
        self.maxEccentricity = maxEccentricity
    }

    public func violates(_ a: FrameAnalysis) -> Bool {
        if let t = maxFWHM, let v = a.stars.fwhmArcsec ?? a.stars.fwhm, v > t { return true }
        if let t = minStars, a.stars.count < t { return true }
        if let t = maxBackground, a.background > t { return true }
        if let t = maxEccentricity, let e = a.stars.eccentricity, e > t { return true }
        return false
    }
}
```

```swift
    public static let rejectFolderName = "_scartati"

    private let trasher: FileTrashing

    public init(trasher: FileTrashing = SystemTrash()) {
        self.trasher = trasher
    }

    public func moveToTrash(_ urls: [URL]) -> TrashReport {
        var report = TrashReport()
        for url in urls {
            do {
                try trasher.trashItem(at: url)
                report.trashed.append(url)
            } catch let error as CocoaError where error.code == .featureUnsupported {
                report.trashUnavailable.append(url)
            } catch {
                report.failed[url] = error.localizedDescription
            }
        }
        return report
    }

    /// Sposta in `_scartati/` accanto a ogni file, senza sovrascrivere ("nome 2.fits", …).
    public func moveToRejectFolder(_ urls: [URL]) -> FileOpReport {
        let fm = FileManager.default
        var report = FileOpReport()
        for url in urls {
            do {
                let folder = url.deletingLastPathComponent().appendingPathComponent(Self.rejectFolderName, isDirectory: true)
                try fm.createDirectory(at: folder, withIntermediateDirectories: true)
                var destination = folder.appendingPathComponent(url.lastPathComponent)
                var n = 2
                while fm.fileExists(atPath: destination.path) {
                    destination = folder.appendingPathComponent(
                        "\(url.deletingPathExtension().lastPathComponent) \(n).\(url.pathExtension)")
                    n += 1
                }
                try fm.moveItem(at: url, to: destination)
                report.done.append(url)
            } catch {
                report.failed[url] = error.localizedDescription
            }
        }
        return report
    }

    public func deletePermanently(_ urls: [URL]) -> FileOpReport {
        var report = FileOpReport()
        for url in urls {
            do {
                try FileManager.default.removeItem(at: url)
                report.done.append(url)
            } catch {
                report.failed[url] = error.localizedDescription
            }
        }
        return report
    }
}
```

```swift
/// Ordine di lavoro: prima il file corrente, poi i vicini alternando avanti/indietro.
public enum AnalysisOrder {
    public static func indices(count: Int, around current: Int?) -> [Int] {
        guard count > 0 else { return [] }
        guard let c = current, (0..<count).contains(c) else { return Array(0..<count) }
        var out = [c]
        var d = 1
        while out.count < count {
            if c + d < count { out.append(c + d) }
            if c - d >= 0 { out.append(c - d) }
            d += 1
        }
        return out
    }

    /// Indici entro `radius` dal corrente, nell'ordine di `indices`; vuoto senza selezione.
    public static func window(count: Int, around current: Int?, radius: Int) -> [Int] {
        guard let c = current, (0..<count).contains(c) else { return [] }
        return indices(count: count, around: c).filter { abs($0 - c) <= radius }
    }
}
```

```swift
import AppKit
import FitsCullCore
import Observation
import SwiftUI

enum FrameStatus: Equatable {
    case pending
    case analyzing
    case done
    case failed(String)
}

struct FrameRecord: Identifiable {
    let id: URL
    /// Posizione cronologica nella cartella (1…n).
    let index: Int
    var file: FrameFile
    var header: FITSHeader?
    var meta: FrameMeta?
    var analysis: FrameAnalysis?
    var status: FrameStatus
    var rejected = false
    var flags: OutlierFlags = []
}

struct Preview: @unchecked Sendable {
    let image: CGImage
    let histogram: DisplayHistogram
}

enum ViewMode: String, CaseIterable {
    case review = "Revisione"
    case session = "Sessione"
}

enum SortKey: String, CaseIterable {
    case date = "Data"
    case fwhm = "FWHM"
    case stars = "Stelle"
    case background = "Fondo"
}

@MainActor
@Observable
final class SessionModel {
    var folder: URL?
    var records: [FrameRecord] = []
    var selection: URL? {
        didSet { if selection != oldValue { selectionChanged() } }
    }
    var typeFilter: ImageType? = .light {
        didSet { refreshSummary(); pump() }
    }
    var sortKey: SortKey = .date
    var viewMode: ViewMode = .review
    var stretchEnabled = true {
        didSet { invalidatePreviews() }
    }
    var linkedChannels = true {
        didSet { invalidatePreviews() }
    }
    var actualSize = false
    var summary: SessionSummary?
    var thresholds = Thresholds()
    var previews: [URL: Preview] = [:]
    var loadingMessage: String?
    var errorMessage: String?

    let concurrency = 3
    let previewRadius = 5
    let cache = MetricsCache(directory: MetricsCache.defaultDirectory())

    @ObservationIgnored private var currentSource: (url: URL, source: PreviewSource)?
    @ObservationIgnored private var analysesInFlight: Set<URL> = []
    @ObservationIgnored private var previewsInFlight: Set<URL> = []
    @ObservationIgnored private var activeJobs = 0
    @ObservationIgnored private var generation = 0
    @ObservationIgnored private var renderVersion = 0

    // MARK: - Viste derivate

    func isVisible(_ record: FrameRecord) -> Bool {
        typeFilter == nil || record.meta == nil || record.meta?.imageType == typeFilter
    }

    var visibleRecords: [FrameRecord] {
        let list = records.filter(isVisible)
        switch sortKey {
        case .date:
            return list
        case .fwhm:
            return list.sorted { ($0.analysis?.stars.fwhm ?? .infinity) < ($1.analysis?.stars.fwhm ?? .infinity) }
        case .stars:
            return list.sorted { ($0.analysis?.stars.count ?? -1) > ($1.analysis?.stars.count ?? -1) }
        case .background:
            return list.sorted { ($0.analysis?.background ?? .infinity) < ($1.analysis?.background ?? .infinity) }
        }
    }

    var selectedRecord: FrameRecord? {
        selection.flatMap { id in records.first { $0.id == id } }
    }

    var rejectedCount: Int { records.filter(\.rejected).count }

    var totalBytes: Int64 { records.reduce(0) { $0 + $1.file.size } }

    var availableTypes: [ImageType] {
        ImageType.allCases.filter { type in records.contains { $0.meta?.imageType == type } }
    }

    // MARK: - Azioni

    func chooseFolder() {
        let panel = NSOpenPanel()
        panel.canChooseDirectories = true
        panel.canChooseFiles = false
        panel.allowsMultipleSelection = false
        panel.prompt = "Apri"
        if panel.runModal() == .OK, let url = panel.url { open(url) }
    }

    func open(_ url: URL) {
        saveCache()
        generation += 1
        let gen = generation
        folder = url
        records = []
        previews = [:]
        currentSource = nil
        analysesInFlight = []
        previewsInFlight = []
        summary = nil
        selection = nil
        loadingMessage = "Lettura intestazioni…"
        Task {
            let result = await Task.detached(priority: .userInitiated) {
                Result(catching: { try FolderScanner.scan(url) })
            }.value
            guard gen == generation else { return }
            loadingMessage = nil
            switch result {
            case .failure(let error):
                errorMessage = UserMessage.from(error)
            case .success(let scanned):
                records = scanned.enumerated().map { i, s in
                    let cached = s.meta == nil ? nil : cache.lookup(s.file)
                    let status: FrameStatus = if let e = s.headerError { .failed(e) } else if cached != nil { .done } else { .pending }
                    return FrameRecord(id: s.file.url, index: i + 1, file: s.file, header: s.header, meta: s.meta,
                                       analysis: cached, status: status)
                }
                if records.isEmpty { errorMessage = "Nessun file FITS (.fit, .fits, .fts) in questa cartella." }
                if let type = typeFilter, !records.contains(where: { $0.meta?.imageType == type }) { typeFilter = nil }
                refreshSummary()
                selection = visibleRecords.first?.id
                pump()
            }
        }
    }

    func move(_ delta: Int) {
        let list = visibleRecords
        guard !list.isEmpty else { return }
        let i = list.firstIndex { $0.id == selection } ?? 0
        selection = list[min(max(i + delta, 0), list.count - 1)].id
    }

    func toggleReject() {
        guard let id = selection else { return }
        update(id) { $0.rejected.toggle() }
    }

    func saveCache() {
        try? cache.save()
    }

    /// Rilegge l'header (es. NAS ricollegato) e rimette il file in coda.
    func retry(_ url: URL) {
        Task {
            let header = await Task.detached(priority: .userInitiated) { try? FITSReader.readHeader(url: url) }.value
            update(url) { record in
                if let header {
                    record.header = header
                    record.meta = FrameMeta(header: header)
                    record.status = .pending
                } else {
                    record.status = .failed("file non leggibile (disco o NAS scollegato?)")
                }
            }
            pump()
        }
    }

    // MARK: - Scheduling dell'analisi

    private enum Job {
        case analyze(URL, withPreview: Bool)
        case preview(URL)
    }

    private func selectionChanged() {
        actualSize = false
        let window = Set(previewWindow())
        previews = previews.filter { window.contains($0.key) }
        if currentSource?.url != selection { currentSource = nil }
        pump()
    }

    private func previewWindow() -> [URL] {
        let list = visibleRecords
        let position = list.firstIndex { $0.id == selection }
        return AnalysisOrder.window(count: list.count, around: position, radius: previewRadius).map { list[$0].id }
    }

    private func analysisQueue() -> [URL] {
        let list = visibleRecords
        let position = list.firstIndex { $0.id == selection }
        let near = AnalysisOrder.indices(count: list.count, around: position).map { list[$0].id }
        return near + records.filter { !isVisible($0) }.map(\.id)
    }

    private func nextJob() -> Job? {
        let lookup = Dictionary(uniqueKeysWithValues: records.map { ($0.id, $0) })
        let window = previewWindow()
        func needsPreview(_ url: URL) -> Bool {
            previews[url] == nil && !previewsInFlight.contains(url) && !analysesInFlight.contains(url)
        }
        if let selected = selection, let record = lookup[selected], needsPreview(selected) {
            switch record.status {
            case .pending: return .analyze(selected, withPreview: true)
            case .done: return .preview(selected)
            default: break
            }
        }
        for url in analysisQueue() where lookup[url]?.status == .pending && !analysesInFlight.contains(url) {
            return .analyze(url, withPreview: window.contains(url) && needsPreview(url))
        }
        for url in window where lookup[url]?.status == .done && needsPreview(url) {
            return .preview(url)
        }
        return nil
    }

    private func pump() {
        while activeJobs < concurrency, let job = nextJob() { start(job) }
        if activeJobs == 0 { saveCache() }
    }

    private func start(_ job: Job) {
        activeJobs += 1
        let gen = generation
        switch job {
        case .analyze(let url, let withPreview):
            analysesInFlight.insert(url)
            if withPreview { previewsInFlight.insert(url) }
            update(url) { $0.status = .analyzing }
            Task {
                let result = await Task.detached(priority: .userInitiated) {
                    Result(catching: { try FrameAnalyzer.analyze(url: url, includePreview: withPreview) })
                }.value
                self.finishAnalysis(url, result, generation: gen)
            }
        case .preview(let url):
            previewsInFlight.insert(url)
            Task {
                let result = await Task.detached(priority: .userInitiated) {
                    Result(catching: { try FrameAnalyzer.preview(url: url) })
                }.value
                self.finishPreview(url, result, generation: gen)
            }
        }
    }

    private func finishAnalysis(_ url: URL, _ result: Result<(analysis: FrameAnalysis, preview: PreviewSource?), Error>,
                                generation gen: Int) {
        activeJobs -= 1
        guard gen == generation else { pump(); return }
        analysesInFlight.remove(url)
        switch result {
        case .success(let output):
            update(url) {
                $0.analysis = output.analysis
                $0.status = .done
            }
            if let file = records.first(where: { $0.id == url })?.file { cache.store(output.analysis, for: file) }
            refreshSummary()
            if let source = output.preview { render(url, source) } else { previewsInFlight.remove(url) }
        case .failure(let error):
            previewsInFlight.remove(url)
            update(url) { $0.status = .failed(UserMessage.from(error)) }
        }
        pump()
    }

    private func finishPreview(_ url: URL, _ result: Result<PreviewSource, Error>, generation gen: Int) {
        activeJobs -= 1
        guard gen == generation else { pump(); return }
        switch result {
        case .success(let source):
            render(url, source)
        case .failure(let error):
            previewsInFlight.remove(url)
            update(url) { $0.status = .failed(UserMessage.from(error)) }
        }
        pump()
    }

    /// Stretch fuori dal main thread; `url` deve essere già in `previewsInFlight`.
    private func render(_ url: URL, _ source: PreviewSource) {
        if url == selection { currentSource = (url, source) }
        let gen = generation, version = renderVersion
        let stretch = stretchEnabled, linked = linkedChannels
        Task {
            let preview = await Task.detached(priority: .userInitiated) {
                SessionModel.makePreview(source, stretch: stretch, linked: linked)
            }.value
            guard gen == generation else { return }
            previewsInFlight.remove(url)
            if version == renderVersion, let preview, previewWindow().contains(url) { previews[url] = preview }
            pump()
        }
    }

    nonisolated static func makePreview(_ source: PreviewSource, stretch: Bool, linked: Bool) -> Preview? {
        let stretched = AutoStretch.stretch(source.image, normalization: source.normalization, linked: linked, enabled: stretch)
        guard let image = AutoStretch.cgImage(stretched) else { return nil }
        return Preview(image: image, histogram: DisplayHistogram(stretched, isColor: source.image.isColor))
    }

    private func invalidatePreviews() {
        renderVersion += 1
        previews = [:]
        if let current = currentSource, current.url == selection, !previewsInFlight.contains(current.url) {
            previewsInFlight.insert(current.url)
            render(current.url, current.source)
        }
        pump()
    }

    // MARK: - Supporto

    func update(_ url: URL, _ change: (inout FrameRecord) -> Void) {
        guard let i = records.firstIndex(where: { $0.id == url }) else { return }
        change(&records[i])
    }

    func refreshSummary() {
        let newSummary = OutlierRules.summary(records.filter(isVisible).compactMap(\.analysis))
        var updated = records
        for i in updated.indices {
            if let s = newSummary, let a = updated[i].analysis {
                updated[i].flags = OutlierRules.flags(for: a, in: s)
            } else {
                updated[i].flags = []
            }
        }
        summary = newSummary
        records = updated
    }
}
```

```swift
import FitsCullCore
import Foundation

enum Format {
    private static let timeFormatter: DateFormatter = {
        let formatter = DateFormatter()
        formatter.dateFormat = "HH:mm"
        return formatter
    }()

    static func time(_ date: Date) -> String { timeFormatter.string(from: date) }

    static func number(_ value: Double?, digits: Int, suffix: String = "") -> String {
        guard let value, value.isFinite else { return "–" }
        return String(format: "%.\(digits)f", value) + suffix
    }

    static func bytes(_ count: Int64) -> String {
        ByteCountFormatter.string(fromByteCount: count, countStyle: .file)
    }

    static func metaLine(_ m: FrameMeta) -> String {
        let parts: [String?] = [m.imageType.label, m.exposure.map { number($0, digits: 0, suffix: " s") },
                                m.gain.map { "gain \($0)" }, m.sensorTemp.map { number($0, digits: 1, suffix: " °C") },
                                m.filter]
        return parts.compactMap { $0 }.joined(separator: " · ")
    }

    static func opticsLine(_ m: FrameMeta) -> String {
        let parts: [String?] = [m.pixelScale.map { number($0, digits: 2, suffix: "″/px") }, m.bayerPattern,
                                m.focalLengthMM.map { number($0, digits: 0, suffix: " mm") }, m.object]
        return parts.compactMap { $0 }.joined(separator: " · ")
    }

    static func flags(_ f: OutlierFlags) -> String {
        var parts: [String] = []
        if f.contains(.background) { parts.append("fondo alto") }
        if f.contains(.stars) { parts.append("poche stelle") }
        if f.contains(.fwhm) { parts.append("FWHM alta") }
        if f.contains(.eccentricity) { parts.append("stelle allungate") }
        return parts.joined(separator: ", ")
    }
}
```

```swift
struct ImageCanvasView: View {
    @Environment(SessionModel.self) private var model

    var body: some View {
        ZStack {
            Color.black
            if let record = model.selectedRecord {
                content(for: record)
                if record.rejected {
                    VStack {
                        HStack {
                            Spacer()
                            Text("SCARTATO")
                                .font(.headline)
                                .padding(6)
                                .background(Color.red.opacity(0.85), in: .rect(cornerRadius: 6))
                                .foregroundStyle(.white)
                        }
                        Spacer()
                    }
                    .padding()
                }
            } else {
                Text("Apri una cartella con ⌘O").foregroundStyle(.secondary)
            }
        }
    }

    @ViewBuilder
    private func content(for record: FrameRecord) -> some View {
        if case .failed(let message) = record.status {
            VStack(spacing: 10) {
                Label(message, systemImage: "exclamationmark.triangle").foregroundStyle(.white)
                Button("Riprova") { model.retry(record.id) }
            }
        } else if let preview = model.previews[record.id] {
            if model.actualSize {
                ScrollView([.horizontal, .vertical]) {
                    Image(decorative: preview.image, scale: NSScreen.main?.backingScaleFactor ?? 2)
                        .interpolation(.none)
                }
            } else {
                Image(decorative: preview.image, scale: 1)
                    .resizable()
                    .interpolation(.medium)
                    .aspectRatio(contentMode: .fit)
            }
        } else {
            ProgressView().controlSize(.large).tint(.white)
        }
    }
}
```

```swift
import FitsCullCore
import SwiftUI

struct HistogramView: View {
    let histogram: DisplayHistogram

    var body: some View {
        Canvas { context, size in
            let colors: [Color] = histogram.channels.count == 3 ? [.red, .green, .blue] : [.gray]
            for (c, bins) in histogram.channels.enumerated() {
                let peak = log(Double((bins.max() ?? 0) + 1))
                guard peak > 0 else { continue }
                var path = Path()
                for (i, count) in bins.enumerated() {
                    let point = CGPoint(x: size.width * CGFloat(i) / 255,
                                        y: size.height * (1 - CGFloat(log(Double(count + 1)) / peak)))
                    if i == 0 { path.move(to: point) } else { path.addLine(to: point) }
                }
                context.stroke(path, with: .color(colors[c].opacity(0.85)), lineWidth: 1)
            }
        }
        .background(Color.black.opacity(0.85), in: .rect(cornerRadius: 4))
    }
}
```

```swift
struct ChartPoint: Identifiable {
    let id: URL
    let x: Date
    let y: Double
    let rejected: Bool
    let outOfThreshold: Bool
}

struct SessionChartsView: View {
    @Environment(SessionModel.self) private var model

    var body: some View {
        HStack(spacing: 0) {
            ScrollView {
                VStack(spacing: 18) {
                    MetricChart(title: "FWHM (″)", points: points { $0.stars.fwhmArcsec ?? $0.stars.fwhm },
                                threshold: model.thresholds.maxFWHM, onPick: open)
                    MetricChart(title: "Stelle", points: points { Double($0.stars.count) },
                                threshold: model.thresholds.minStars.map(Double.init), onPick: open)
                    MetricChart(title: "Fondo (ADU)", points: points { $0.background },
                                threshold: model.thresholds.maxBackground, onPick: open)
                    MetricChart(title: "Eccentricità", points: points { $0.stars.eccentricity },
                                threshold: model.thresholds.maxEccentricity, onPick: open)
                }
                .padding()
            }
            Divider()
            ThresholdForm().frame(width: 250)
        }
    }

    private func points(_ value: (FrameAnalysis) -> Double?) -> [ChartPoint] {
        model.visibleRecords.compactMap { r in
            guard let a = r.analysis, let y = value(a), y.isFinite else { return nil }
            return ChartPoint(id: r.id, x: r.meta?.dateObs ?? Date(timeIntervalSinceReferenceDate: Double(r.index) * 60),
                              y: y, rejected: r.rejected, outOfThreshold: model.thresholds.violates(a))
        }
    }

    private func open(_ id: URL) {
        model.selection = id
        model.viewMode = .review
    }
}

struct MetricChart: View {
    let title: String
    let points: [ChartPoint]
    let threshold: Double?
    let onPick: (URL) -> Void
    @State private var selectedX: Date?

    var body: some View {
        VStack(alignment: .leading, spacing: 4) {
            Text(title).font(.headline)
            Chart {
                ForEach(points) { p in
                    PointMark(x: .value("Ora", p.x), y: .value(title, p.y))
                        .foregroundStyle(p.rejected ? Color.red : (p.outOfThreshold ? Color.orange : Color.accentColor))
                        .symbolSize(p.rejected ? 50 : 30)
                }
                if let threshold {
                    RuleMark(y: .value("Soglia", threshold))
                        .foregroundStyle(.orange)
                        .lineStyle(StrokeStyle(lineWidth: 1, dash: [4, 3]))
                }
            }
            .chartXAxis { AxisMarks(values: .automatic) { _ in AxisGridLine(); AxisValueLabel(format: .dateTime.hour().minute()) } }
            .chartXSelection(value: $selectedX)
            .onChange(of: selectedX) { _, x in
                guard let x, let nearest = points.min(by: { abs($0.x.timeIntervalSince(x)) < abs($1.x.timeIntervalSince(x)) })
                else { return }
                onPick(nearest.id)
            }
            .frame(height: 140)
        }
    }
}
```

```swift
    // MARK: - Cestino

    var rejectedBytes: Int64 { records.filter(\.rejected).reduce(0) { $0 + $1.file.size } }

    var trashFallbackBytes: Int64 {
        let pending = Set(trashFallback)
        return records.filter { pending.contains($0.id) }.reduce(0) { $0 + $1.file.size }
    }

    func requestTrash() {
        if rejectedCount > 0 { confirmTrash = true }
    }

    func performTrash() {
        let report = TrashService().moveToTrash(records.filter(\.rejected).map(\.id))
        removeRecords(report.trashed)
        trashFallback = report.trashUnavailable + report.failed.keys.sorted { $0.path < $1.path }
        trashFallbackReason = report.failed.values.first
    }

    func moveFallbackToRejectFolder() {
        let report = TrashService().moveToRejectFolder(trashFallback)
        removeRecords(report.done)
        trashFallback = []
        if let reason = report.failed.values.first {
            errorMessage = "Non è stato possibile spostare \(report.failed.count) file: \(reason)"
        }
    }

    func deleteFallbackPermanently() {
        let report = TrashService().deletePermanently(trashFallback)
        removeRecords(report.done)
        trashFallback = []
        if let reason = report.failed.values.first {
            errorMessage = "Non è stato possibile eliminare \(report.failed.count) file: \(reason)"
        }
    }

    private func removeRecords(_ urls: [URL]) {
        guard !urls.isEmpty else { return }
        let gone = Set(urls)
        let list = visibleRecords
        var next = selection
        if let selected = selection, gone.contains(selected), let i = list.firstIndex(where: { $0.id == selected }) {
            next = list[i...].first { !gone.contains($0.id) }?.id ?? list[..<i].last { !gone.contains($0.id) }?.id
        }
        records.removeAll { gone.contains($0.id) }
        for url in urls {
            cache.remove(url)
            previews[url] = nil
        }
        saveCache()
        selection = next
        refreshSummary()
    }
```

```bash
#!/bin/zsh
# Genera build/AppIcon.icns da scripts/make-icon.swift.
set -euo pipefail
cd "$(dirname "$0")/.."
rm -rf build/AppIcon.iconset
mkdir -p build/AppIcon.iconset
swift scripts/make-icon.swift build/icon-1024.png
for s in 16 32 128 256 512; do
  sips -z $s $s build/icon-1024.png --out build/AppIcon.iconset/icon_${s}x${s}.png >/dev/null
  d=$((s * 2))
  sips -z $d $d build/icon-1024.png --out build/AppIcon.iconset/icon_${s}x${s}@2x.png >/dev/null
done
iconutil -c icns build/AppIcon.iconset -o build/AppIcon.icns
echo "Icona: build/AppIcon.icns"
```

`scripts/build-app.sh`:
```bash
#!/bin/zsh
# Compila in release, crea FitsCull.app, firma ad-hoc e installa nella destinazione.
set -euo pipefail
cd "$(dirname "$0")/.."
DEST="${1:-/Volumes/Dati2/applicazioni}"
[[ -d "$DEST" ]] || { echo "Destinazione non trovata: $DEST" >&2; exit 1; }

swift build -c release --product FitsCull
BIN="$(swift build -c release --show-bin-path)/FitsCull"
[[ -f build/AppIcon.icns ]] || scripts/make-icon.sh

APP=build/FitsCull.app
rm -rf "$APP"
mkdir -p "$APP/Contents/MacOS" "$APP/Contents/Resources"
cp "$BIN" "$APP/Contents/MacOS/FitsCull"
cp Resources/Info.plist "$APP/Contents/Info.plist"
cp build/AppIcon.icns "$APP/Contents/Resources/AppIcon.icns"
codesign --force --sign - "$APP"

rm -rf "$DEST/FitsCull.app"
cp -R "$APP" "$DEST/"
echo "Installata: $DEST/FitsCull.app"
```
